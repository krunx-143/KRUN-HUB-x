local Remotes = ReplicatedStorage:WaitForChild("Remotes", 10) or ReplicatedStorage:FindFirstChild("Remotes")

local function safeFire(remoteName, ...)
    if Remotes and Remotes:FindFirstChild(remoteName) then
        pcall(function(...)
            Remotes[remoteName]:FireServer(...)
        end, ...)
    end
end

local function safeInvoke(remoteName, ...)
    if Remotes and Remotes:FindFirstChild(remoteName) then
        local success, result = pcall(function(...)
            return Remotes[remoteName]:InvokeServer(...)
        end, ...)
        if success then return result end
    end
    return nil
end

-- ==========================================
--  ระบบ BYPASS TELEPORT (กันดีด / กันตกแมพ)
-- ==========================================
local function bypassTeleport(targetCFrame)
    local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChild("Humanoid")
    if not root or not hum then return end

    root.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
    root.AssemblyAngularVelocity = Vector3.new(0, 0, 0)

    safeFire("MovementSync", targetCFrame.Position)

    local originalState = hum:GetState()
    hum:ChangeState(Enum.HumanoidStateType.Physics)
    task.wait(0.03)
    root.CFrame = targetCFrame + Vector3.new(0, 2.5, 0)
    task.wait(0.03)
    hum:ChangeState(originalState)
    root.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
end

local function registerText(obj, key)
    textRegistry[obj] = key
    if LangDict[key] and LangDict[key][currentLang] then
        obj.Text = LangDict[key][currentLang]
    end
end

local function toggleLanguage()
    currentLang = (currentLang == "TH") and "EN" or "TH"
    for obj, key in pairs(textRegistry) do
        if obj and obj.Parent and LangDict[key] then
            obj.Text = LangDict[key][currentLang]
        end
    end
end

-- ==========================================
--  สีตัวหนังสือ แดงไล่ดำ (Red to Black)
-- ==========================================
local function applyRedToBlackGradient(textLabel)
    local grad = Instance.new("UIGradient")
    grad.Color = ColorSequence.new{
        ColorSequenceKeypoint.new(0, Color3.fromRGB(255, 45, 45)),
        ColorSequenceKeypoint.new(1, Color3.fromRGB(20, 5, 5))
    }
    grad.Rotation = 90
    grad.Parent = textLabel
    return grad
end

            data.Page.Visible = (name == pageName)
        end
        for name, btn in pairs(TabButtons) do
            if name == pageName then
                btn.BackgroundTransparency = 0
                btn.BackgroundColor3 = Color3.fromRGB(26, 30, 44)
                btn.TextColor3 = Color3.fromRGB(255, 255, 255)
            else
                btn.BackgroundTransparency = 1
                btn.TextColor3 = Color3.fromRGB(130, 130, 150)
            end
        end
    end)
end
