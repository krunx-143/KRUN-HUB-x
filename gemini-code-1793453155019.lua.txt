-- =========================================================
-- 🛠️ SAFE REMOTE WRAPPERS
-- =========================================================
local function GetAdminKey()
    if State.CachedAdminKey then return State.CachedAdminKey end
    local r = Remotes and Remotes:FindFirstChild("GetAdminKey")
    if r and r:IsA("RemoteFunction") then
        local ok, res = pcall(function() return r:InvokeServer() end)
        if ok and res then
            State.CachedAdminKey = res
            return res
        end
    end
    return nil
end

local function FireRemote(name, ...)
    local r = Remotes and Remotes:FindFirstChild(name)
    if not r then return end
    local args = { ... }
    task.spawn(function()
        pcall(function()
            if r:IsA("RemoteEvent") or r:IsA("UnreliableRemoteEvent") then
                r:FireServer(unpack(args))
            elseif r:IsA("RemoteFunction") then
                r:InvokeServer(unpack(args))
            end
        end)
    end)
end

local function InvokeRemote(name, ...)
    local r = Remotes and Remotes:FindFirstChild(name)
    if not r then return nil end
    local args = { ... }
    local ok, res = pcall(function()
        if r:IsA("RemoteFunction") then
            return r:InvokeServer(unpack(args))
        elseif r:IsA("RemoteEvent") then
            r:FireServer(unpack(args))
        end
    end)
    return ok and res or nil
end

-- Multi-signature caller to cover varying server argument checks
local function SmartMoney(amount)
    local key = GetAdminKey()
    FireRemote("GiveMoney", amount)
    FireRemote("GiveMoney", LocalPlayer, amount)
    if key then FireRemote("GiveMoney", amount, key) end

    FireRemote("SetCash", amount)
    FireRemote("SetCash", LocalPlayer, amount)
    if key then FireRemote("SetCash", amount, key) end
end

local function SmartGiveItem(itemName, count)
    local key = GetAdminKey()
    FireRemote("AdminGiveItem", itemName, count)
    FireRemote("AdminGiveItem", itemName)
    FireRemote("AdminGiveItem", { Item = itemName, Amount = count })
    if key then FireRemote("AdminGiveItem", key, itemName, count) end
end

local function SmartDupe(itemName, count)
    FireRemote("DupeItem", itemName, count)
    FireRemote("DupeItem", itemName)
    FireRemote("DupeItem")
    local char = LocalPlayer.Character
    local tool = char and char:FindFirstChildOfClass("Tool")
    if tool then
        FireRemote("DupeItem", tool.Name, count)
        FireRemote("DupeItem", tool)
    end
end

local function FireAllProximityPrompts()
    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root or not fireproximityprompt then return end

    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("ProximityPrompt") and obj.Enabled then
            local part = obj.Parent
            local pos = nil
            if part:IsA("BasePart") then
                pos = part.Position
            elseif part:IsA("Attachment") then
                pos = part.WorldPosition
            elseif part:IsA("Model") and part.PrimaryPart then
                pos = part.PrimaryPart.Position
            end

            if pos and (pos - root.Position).Magnitude <= (obj.MaxActivationDistance + 15) then
                pcall(function()
                    fireproximityprompt(obj)
                end)
            end
        end
    end
end

    local char = LocalPlayer.Character
    if not char then return end

    if State.EnableSpeed then
        local hum = char:FindFirstChildOfClass("Humanoid")

            end
        end
